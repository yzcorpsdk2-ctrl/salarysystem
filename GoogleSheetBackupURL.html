/** * ============================================================ * Google Sheet Backup URL - 独立后端 (多 Sheet 分组版) * ============================================================ * 核心逻辑： * - 每个分组对应一个独立的 Sheet 工作表 * - 分组名 = Sheet 名（已做安全化处理：空格/特殊字符转下划线） * - 前端下拉框动态从后端获取分组列表 * * 部署： * 1. 绑定到你的 Google Sheet * 2. 粘贴代码 * 3. 运行 setupDefaultGroups() 创建默认分组（可选） * 4. 部署为 Web 应用（执行身份：我 / 访问：任何人） * ============================================================ */ const HEADERS = ['ID', 'Group', 'Name', 'URL', 'Timestamp']; // 默认分组（首次 setup 时创建） const DEFAULT_GROUPS = ['Google Sheet', 'App Script', 'Dashboard', 'API', 'Lain-lain']; /* ============================================================ GET 请求 - action=getGroups → 返回所有分组（Sheet 名） - action=getBackupUrls → 返回指定分组的 URL 列表（参数 group） - action=getAllBackupUrls → 返回所有分组的所有记录（可选） ============================================================ */ function doGet(e) { try { const params = (e && e.parameter) ? e.parameter : {}; const action = (params.action || '').trim(); if (action === 'getGroups') { return jsonResponse({ status: 'success', data: getGroups() }); } if (action === 'getBackupUrls') { const group = (params.group || '').trim(); const data = getAllBackupUrls(group); return jsonResponse({ status: 'success', data: data, group: group }); } if (action === 'getAllBackupUrls') { const groups = getGroups(); const all = {}; groups.forEach(g => { all[g] = getAllBackupUrls(g); }); return jsonResponse({ status: 'success', data: all }); } return jsonResponse({ status: 'success', message: 'Backup URL API is running.', availableActions: ['getGroups', 'getBackupUrls', 'getAllBackupUrls'] }); } catch (err) { return jsonResponse({ status: 'error', message: err.message }); } } /* ============================================================ POST 请求 - action=saveBackupUrl → 新增（必须带 group） - action=deleteBackupUrl → 删除（必须带 group + id） - action=createGroup → 新建分组（创建新 Sheet） - action=deleteGroup → 删除分组（删除 Sheet，谨慎！） - action=renameGroup → 重命名分组（重命名 Sheet） ============================================================ */ function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return jsonResponse({ status: 'error', message: 'Tiada data diterima' }); } const body = JSON.parse(e.postData.contents); const action = (body.action || '').trim(); const data = body.data || {}; if (action === 'saveBackupUrl') return jsonResponse(saveBackupUrl(data)); if (action === 'deleteBackupUrl') return jsonResponse(deleteBackupUrl(data)); if (action === 'createGroup') return jsonResponse(createGroup(data)); if (action === 'deleteGroup') return jsonResponse(deleteGroup(data)); if (action === 'renameGroup') return jsonResponse(renameGroup(data)); return jsonResponse({ status: 'error', message: 'Unknown action: ' + action }); } catch (err) { return jsonResponse({ status: 'error', message: err.message }); } } function doOptions(e) { return ContentService.createTextOutput('').setMimeType(ContentService.MimeType.TEXT); } /* ============================================================ 工具函数 ============================================================ */ function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /** * 分组名 → 安全 Sheet 名 * Google Sheet 名限制：不能含 : \ / ? * [ ] * 这里额外把空格转为下划线，方便识别 */ function toSheetName(group) { let name = String(group || '').trim(); if (!name) name = 'Lain-lain'; // 移除非法字符 name = name.replace(/[:\\\/\?\*\[\]]/g, '_'); // 限制长度（Sheet 名最长 100 字符） if (name.length > 80) name = name.substring(0, 80); return name; } /** * 安全 Sheet 名 → 显示分组名（下划线转回空格，可选） * 实际使用时保留原名即可，这里只是给前端一个参考 */ /* ============================================================ 获取所有分组（即所有 Sheet 名，排除系统表） ============================================================ */ function getGroups() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheets = ss.getSheets(); const groups = []; sheets.forEach(sheet => { const name = sheet.getName(); // 排除隐藏工作表（如需保留系统表可加判断） if (sheet.isSheetHidden()) return; // 排除空名 if (!name) return; groups.push(name); }); return groups; } /* ============================================================ 获取（或自动创建）指定分组的 Sheet ============================================================ */ function getSheetByGroup(group) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = toSheetName(group); let sheet = ss.getSheetByName(sheetName); if (!sheet) { // 自动创建 sheet = ss.insertSheet(sheetName); sheet.appendRow(HEADERS); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold'); sheet.setFrozenRows(1); return sheet; } // 空表：写表头 if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold'); sheet.setFrozenRows(1); return sheet; } // 校验表头 const headerRange = sheet.getRange(1, 1, 1, HEADERS.length); const currentHeaders = headerRange.getValues()[0].map(v => String(v || '').trim()); const needFix = HEADERS.some((h, idx) => currentHeaders[idx] !== h); if (needFix) { headerRange.setValues([HEADERS]); headerRange.setFontWeight('bold'); } return sheet; } /* ============================================================ 读取指定分组的所有 URL ============================================================ */ function getAllBackupUrls(group) { const sheetName = toSheetName(group); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) return []; // 分组不存在 → 空数组 const lastRow = sheet.getLastRow(); if (lastRow <= 1) return []; const values = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues(); return values .filter(row => row[0] !== '' || row[3] !== '') .map(row => ({ ID: String(row[0] || ''), Group: String(row[1] || ''), Name: String(row[2] || ''), URL: String(row[3] || ''), Timestamp: row[4] ? new Date(row[4]).toISOString() : '' })); } /* ============================================================ 新增 URL（到指定分组） ============================================================ */ function saveBackupUrl(data) { if (!data) throw new Error('Tiada data diterima'); const group = String(data.group || '').trim(); const name = String(data.name || '').trim(); const url = String(data.url || '').trim(); if (!group) throw new Error('Group diperlukan'); if (!name) throw new Error('Nama tidak boleh kosong'); if (!url) throw new Error('URL tidak boleh kosong'); if (!/^https?:\/\//i.test(url)) { throw new Error('URL mesti bermula dengan http:// atau https://'); } const sheet = getSheetByGroup(group); // 唯一 ID const existingRows = getAllBackupUrls(group); const existingIds = existingRows.map(r => r.ID); let id = String(Date.now()); let counter = 0; while (existingIds.indexOf(id) !== -1) { id = String(Date.now()) + '-' + (++counter); } const timestamp = new Date().toISOString(); sheet.appendRow([id, group, name, url, timestamp]); return { status: 'success', message: 'URL berjaya disimpan ke ' + toSheetName(group), id: id, group: group, sheet: toSheetName(group) }; } /* ============================================================ 删除 URL（从指定分组） ============================================================ */ function deleteBackupUrl(data) { if (!data) throw new Error('Tiada data diterima'); const group = String(data.group || '').trim(); const targetId = String(data.id || '').trim(); const targetTs = String(data.timestamp || '').trim(); if (!group) throw new Error('Group diperlukan'); if (!targetId && !targetTs) throw new Error('ID atau Timestamp diperlukan'); const sheetName = toSheetName(group); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) throw new Error('Sheet tidak dijumpai: ' + sheetName); const lastRow = sheet.getLastRow(); if (lastRow <= 1) throw new Error('Tiada data untuk dipadam'); const values = sheet.getRange(2, 1, lastRow - 1, HEADERS.length).getValues(); let rowToDelete = -1; if (targetId) { for (let i = 0; i < values.length; i++) { if (String(values[i][0] || '') === targetId) { rowToDelete = i + 2; break; } } } if (rowToDelete === -1 && targetTs) { for (let i = 0; i < values.length; i++) { const rowTs = values[i][4] ? new Date(values[i][4]).toISOString() : ''; if (rowTs === targetTs) { rowToDelete = i + 2; break; } } } if (rowToDelete === -1) throw new Error('Rekod tidak dijumpai'); sheet.deleteRow(rowToDelete); return { status: 'success', message: 'URL berjaya dipadam dari ' + sheetName }; } /* ============================================================ 新建分组（创建新 Sheet） ============================================================ */ function createGroup(data) { const group = String(data.group || '').trim(); if (!group) throw new Error('Nama group diperlukan'); const sheetName = toSheetName(group); const ss = SpreadsheetApp.getActiveSpreadsheet(); if (ss.getSheetByName(sheetName)) { throw new Error('Group sudah wujud: ' + sheetName); } const sheet = ss.insertSheet(sheetName); sheet.appendRow(HEADERS); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold'); sheet.setFrozenRows(1); return { status: 'success', message: 'Group berjaya dicipta: ' + sheetName, group: sheetName }; } /* ============================================================ 删除分组（删除 Sheet，谨慎使用！） ============================================================ */ function deleteGroup(data) { const group = String(data.group || '').trim(); if (!group) throw new Error('Nama group diperlukan'); const sheetName = toSheetName(group); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(sheetName); if (!sheet) throw new Error('Sheet tidak dijumpai: ' + sheetName); // 安全检查：不允许删除仅剩的 Sheet if (ss.getSheets().length <= 1) { throw new Error('Tidak boleh padam sheet terakhir'); } ss.deleteSheet(sheet); return { status: 'success', message: 'Group berjaya dipadam: ' + sheetName }; } /* ============================================================ 重命名分组（重命名 Sheet） ============================================================ */ function renameGroup(data) { const oldName = String(data.oldGroup || '').trim(); const newName = String(data.newGroup || '').trim(); if (!oldName || !newName) throw new Error('oldGroup dan newGroup diperlukan'); const oldSheetName = toSheetName(oldName); const newSheetName = toSheetName(newName); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName(oldSheetName); if (!sheet) throw new Error('Sheet tidak dijumpai: ' + oldSheetName); if (ss.getSheetByName(newSheetName)) { throw new Error('Nama group baru sudah wujud: ' + newSheetName); } sheet.setName(newSheetName); // 可选：更新旧数据中的 Group 列（保持一致） const lastRow = sheet.getLastRow(); if (lastRow > 1) { const groupRange = sheet.getRange(2, 2, lastRow - 1, 1); const groupValues = groupRange.getValues().map(row => [newName]); groupRange.setValues(groupValues); } return { status: 'success', message: 'Group berjaya dinamakan semula: ' + newSheetName, group: newSheetName }; } /* ============================================================ 首次运行：创建默认分组 （在 Apps Script 编辑器手动运行一次） ============================================================ */ function setupDefaultGroups() { const ss = SpreadsheetApp.getActiveSpreadsheet(); DEFAULT_GROUPS.forEach(g => { const sheetName = toSheetName(g); if (!ss.getSheetByName(sheetName)) { const sheet = ss.insertSheet(sheetName); sheet.appendRow(HEADERS); sheet.getRange(1, 1, 1, HEADERS.length).setFontWeight('bold'); sheet.setFrozenRows(1); Logger.log('创建分组: ' + sheetName); } else { Logger.log('已存在: ' + sheetName); } }); Logger.log('完成！当前所有分组: ' + getGroups().join(', ')); } /* ============================================================ 测试函数 ============================================================ */ function testApi() { Logger.log('Groups: ' + JSON.stringify(getGroups())); const saved = saveBackupUrl({ group: 'Google Sheet', name: 'Test ' + new Date().toISOString(), url: 'https://example.com' }); Logger.log('Saved: ' + JSON.stringify(saved)); const list = getAllBackupUrls('Google Sheet'); Logger.log('List: ' + JSON.stringify(list)); if (saved && saved.id) { const del = deleteBackupUrl({ group: 'Google Sheet', id: saved.id }); Logger.log('Deleted: ' + JSON.stringify(del)); } }
